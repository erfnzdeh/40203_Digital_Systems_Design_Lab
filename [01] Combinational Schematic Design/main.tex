	\section*{۱.۱ مقدمه و اهداف}
	هدف آزمایش، آشنایی با طراحی شماتیک در ابزارهای \lr{CAD} و پیاده‌سازی مدارهای ترکیبی روی \lr{FPGA} است. مطابق دستور کار، مدارهای تشخیص مضرب ۳ و مضرب ۱۱ برای یک ورودی چهاررقمی \lr{BCD} تنها با گیت‌های پایه طراحی می‌شوند.

	\section*{۱.۲ طراحی مدار}
	\subsection*{الف) تشخیص مضرب ۳}
	برای عدد چهاررقمی \lr{BCD} به‌صورت
	\[
		N=1000A+100B+10C+D
	\]
	داریم \(10\equiv1\pmod 3\)، در نتیجه
	\[
		N\bmod3=(A+B+C+D)\bmod3.
	\]
	برای ساخت جمع‌کننده‌های موردنیاز، ابتدا نیم‌جمع‌کننده با روابط
	\(S=A\oplus B\) و \(C=A B\) از گیت‌های پایه ساخته شد. سپس دو نیم‌جمع‌کننده و یک گیت \lr{OR} یک جمع‌کننده‌ی کامل را تشکیل دادند و چهار نمونه‌ی جمع‌کننده‌ی کامل نیز به‌صورت \lr{ripple-carry} به یک جمع‌کننده‌ی چهار بیتی متصل شدند.

	\shot{screenshots/Div3/1.png}{پیاده‌سازی نیم‌جمع‌کننده با گیت‌های پایه}
	\shot{screenshots/Div3/2.png}{ساخت جمع‌کننده‌ی کامل از دو نیم‌جمع‌کننده}
	\shot{screenshots/Div3/3.png}{جمع‌کننده‌ی چهار بیتی با اتصال آبشاری چهار جمع‌کننده‌ی کامل}
	\shot{screenshots/Div3/4.png}{شبیه‌سازی جمع‌کننده‌ی چهار بیتی و مشاهده‌ی جمع و نقلی}

	بلوک \lr{rem} با منطق ترکیبی، باقیمانده‌ی یک رقم \lr{BCD} بر ۳ را به‌صورت یک مقدار دو بیتی تولید می‌کند. شکل موج مستقل این بلوک، تکرار دنباله‌ی \(0,1,2\) را برای ورودی‌های ده‌گانه نشان می‌دهد. در مدار \lr{Div3} باقیمانده‌ی چهار رقم ابتدا محاسبه، به کمک سه جمع‌کننده با هم جمع و دوباره بر ۳ تقلیل داده می‌شود. خروجی \lr{div3} تنها هنگامی یک است که دو بیت باقیمانده‌ی نهایی صفر باشند.

	\shot{screenshots/Div3/5.png}{منطق گیتی بلوک محاسبه‌ی باقیمانده بر ۳}
	\shot{screenshots/Div3/7.png}{شبیه‌سازی بلوک باقیمانده برای رقم‌های صفر تا ۹}
	\shot{screenshots/Div3/8.png}{شماتیک سلسله‌مراتبی آشکارساز مضرب ۳ چهاررقمی}

	\subsection*{ب) تشخیص مضرب ۱۱}
	از آنجا که \(10\equiv-1\pmod {11}\)، شرط بخش‌پذیری عدد چهاررقمی برابر است با
	\[
		N\bmod11=(D-C+B-A)\bmod11=0,
	\]
	یا به‌صورت معادل، برابری دو جمع متناوب \(A+C\) و \(B+D\) با درنظرگرفتن حالت‌های هم‌نهشت بر ۱۱.

	برای انجام مقایسه، مقایسه‌کننده‌ی یک‌بیتی با سه خروجی بزرگ‌تر، برابر و کوچک‌تر ساخته شد. دو نمونه از آن یک مقایسه‌کننده‌ی دو بیتی و دو نمونه‌ی مقایسه‌کننده‌ی دوبیتی یک مقایسه‌کننده‌ی چهار بیتی را تشکیل می‌دهند؛ در هر سطح، نتیجه‌ی بیت پرارزش بر نتیجه‌ی بیت کم‌ارزش اولویت دارد. یک مالتی‌پلکسر چهار بیتی نیز با چهار مالتی‌پلکسر گیتی یک‌بیتی پیاده‌سازی شد.

	\shot{screenshots/Div11/1.png}{مقایسه‌کننده‌ی یک‌بیتی با خروجی‌های بزرگ‌تر، برابر و کوچک‌تر}
	\shot{screenshots/Div11/2.png}{ساخت مقایسه‌کننده‌ی دوبیتی از دو مقایسه‌کننده‌ی یک‌بیتی}
	\shot{screenshots/Div11/3.png}{ساخت مقایسه‌کننده‌ی چهاربیتی از دو مقایسه‌کننده‌ی دوبیتی}
	\shot{screenshots/Div11/4.png}{شبیه‌سازی حالت‌های بزرگ‌تر، کوچک‌تر و برابر مقایسه‌کننده}
	\shot{screenshots/Div11/5.png}{مالتی‌پلکسر چهار بیتی ساخته‌شده از گیت‌های پایه}

	در مدار نهایی \lr{Div11}، دو جفت رقم متناوب توسط جمع‌کننده‌های چهار بیتی جمع می‌شوند. شبکه‌ی جمع‌کننده‌ها، مالتی‌پلکسرها و مقایسه‌کننده‌ها حالت نقلی و حالت‌های معادل پیمانه‌ای را انتخاب می‌کند و خروجی برابری آخرین مقایسه‌کننده به‌عنوان \lr{Div11} استفاده می‌شود.

	\shot{screenshots/Div11/6.png}{شماتیک سلسله‌مراتبی آشکارساز مضرب ۱۱}

	\subsection*{ج) مدار نهایی}
	دو آشکارساز مستقل در پیمانه‌ی \lr{final} قرار گرفتند و چهار رقم \lr{BCD} مشترک به هر دو اعمال شد. بنابراین خروجی‌های \lr{is\_Div3} و \lr{is\_Div11} می‌توانند هم‌زمان و بدون اثرگذاری بر یکدیگر وضعیت بخش‌پذیری عدد ورودی را گزارش کنند.

	\shot{screenshots/Final/1.png}{اتصال آشکارسازهای مضرب ۳ و مضرب ۱۱ در مدار نهایی}

	\section*{۱.۳ شبیه‌سازی و ارزیابی}
	در شبیه‌سازی \lr{Div3} چند عدد چهاررقمی اعمال شده است. برای نمونه، ارقام \(1,2,3,6\) عدد ۱۲۳۶ را می‌سازند و چون جمع رقم‌ها \(12\) است، خروجی \lr{div3} یک می‌شود. برای ۲۳۴۱ با جمع رقم‌های \(10\) خروجی صفر، برای ۳۱۰۵ با جمع رقم‌های \(9\) خروجی یک و برای ۵۰۳۸ با جمع رقم‌های \(16\) خروجی دوباره صفر است. این رفتار با معیار نظری بخش‌پذیری بر ۳ سازگار است.

	\shot{screenshots/Div3/9.png}{شبیه‌سازی آشکارساز مضرب ۳ برای چند عدد چهاررقمی}

	شکل موج \lr{Div11} نیز چند بردار برابر و نابرابر از نظر دو جمع متناوب را آزمایش می‌کند. خروجی \lr{Div11} برای حالت‌های منطبق با شرط بخش‌پذیری فعال و با نقض این شرط غیرفعال می‌شود؛ تغییر خروجی میان بردارهای نمایش‌داده‌شده، کارکرد هر دو مسیر مدار را نشان می‌دهد.

	\shot{screenshots/Div11/7.png}{شبیه‌سازی آشکارساز مضرب ۱۱ برای بردارهای آزمون}

	در آزمون مدار نهایی، هر چهار رقم به هر دو زیرمدار داده شده‌اند. شکل موج نشان می‌دهد که دو خروجی مستقل‌اند: برای ۳۶۶۳ هر دو شرط برقرارند؛ برای ۱۱۱۲ هیچ‌کدام برقرار نیست؛ برای ۵۷۶۳ فقط شرط مضرب ۳ و برای ۶۱۲۷ فقط شرط مضرب ۱۱ برقرار است. این چهار حالت، ترکیب‌های ممکن دو خروجی را پوشش می‌دهند.

	\shot{screenshots/Final/3.png}{شبیه‌سازی یکپارچه و پوشش چهار حالت خروجی دو آشکارساز}

	\section*{۱.۴ سنتز و پیاده‌سازی نرم‌افزاری}
	نمای \lr{RTL} مدار مضرب ۳ مسیر مورد انتظارِ محاسبه‌ی باقیمانده‌ی هر رقم، جمع نتایج و آشکارسازی باقیمانده‌ی صفر را حفظ کرده است. نماهای پس از \lr{mapping} و پس از \lr{fitting} نشان می‌دهند که ساختار گیتی به منابع واقعی قطعه نگاشت و مسیریابی شده است.

	در مدار مضرب ۱۱ نیز نمای \lr{RTL} زنجیره‌ی جمع‌کننده‌ها، مالتی‌پلکسرها و مقایسه‌کننده‌ی نهایی را نشان می‌دهد. نماهای فناوری پس از نگاشت و جانمایی، پیاده‌سازی همین مسیر ترکیبی را روی منابع تراشه تأیید می‌کنند.

	نمای \lr{RTL} مدار نهایی دو بلوک \lr{Div3} و \lr{Div11} را با ورودی‌های مشترک نشان می‌دهد. در نماهای پس از نگاشت و پس از \lr{fitting} نیز ۱۶ پایه‌ی ورودی مربوط به چهار رقم و دو پایه‌ی خروجی دیده می‌شوند.

	\section*{۱.۵ نتایج}
	\begin{itemize}
		\item زیرمدارهای نیم‌جمع‌کننده، جمع‌کننده‌ی کامل، جمع‌کننده‌ی چهار بیتی، مقایسه‌کننده‌ها و مالتی‌پلکسر تنها با شماتیک و گیت‌های پایه ساخته شدند.
		\item معیار جمع رقم‌ها برای تشخیص مضرب ۳ به‌صورت سلسله‌مراتبی پیاده‌سازی و در شبیه‌سازی تأیید شد.
		\item معیار اختلاف جمع رقم‌های متناوب برای تشخیص مضرب ۱۱ با جمع‌کننده، مالتی‌پلکسر و مقایسه‌کننده پیاده‌سازی شد.
		\item شبیه‌سازی مدار نهایی هر چهار ترکیب ممکن خروجی‌های \lr{is\_Div3} و \lr{is\_Div11} را نشان داد.
		\item هر سه طرح بدون استفاده از ثبات و با مصرف کمتر از یک درصد عناصر منطقی قطعه سنتز شدند.
	\end{itemize}

	\section*{نتیجه‌گیری}
	در این آزمایش دو مدار ترکیبی تشخیص بخش‌پذیری برای ورودی چهاررقمی \lr{BCD} با طراحی شماتیک ساخته شدند. شکستن طرح به بلوک‌های کوچک و قابل‌آزمون، از نیم‌جمع‌کننده و مقایسه‌کننده‌ی یک‌بیتی تا آشکارسازهای کامل، بررسی مرحله‌به‌مرحله‌ی عملکرد را ممکن کرد. شکل موج‌ها با قواعد ریاضی بخش‌پذیری بر ۳ و ۱۱ سازگارند و گزارش‌های سنتز و نماهای پس از نگاشت و جانمایی نیز پیاده‌سازی موفق و کم‌هزینه‌ی طرح روی \lr{Cyclone IV GX} را تأیید می‌کنند.
