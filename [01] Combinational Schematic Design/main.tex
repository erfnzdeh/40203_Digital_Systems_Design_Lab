	\section*{۱.۱ مقدمه و اهداف}
	هدف آزمایش، آشنایی با طراحی طرح‌واره‌ای به کمک رایانه و پیاده‌سازی مدارهای ترکیبی روی آرایهٔ دروازه‌ای برنامه‌پذیر میدانی است. مطابق دستورکار، مدارهای تشخیص مضرب ۳ و مضرب ۱۱ برای یک ورودی چهاررقمی با رمزگذاری دهدهیِ دودویی‌شده، تنها با دروازه‌های پایه طراحی می‌شوند.

	\section*{۱.۲ طراحی مدار}
	\subsection*{الف) تشخیص مضرب ۳}
	برای عدد چهاررقمی با رمزگذاری دهدهیِ دودویی‌شده به‌صورت
	\[
		N=1000A+100B+10C+D
	\]
	داریم \(10\equiv1\pmod 3\)، در نتیجه
	\[
		N\bmod3=(A+B+C+D)\bmod3.
	\]
	برای ساخت جمع‌کننده‌های مورد نیاز، ابتدا نیم‌جمع‌کننده با روابط
	\(S=A\oplus B\) و \(C=A B\) ساخته شد. برای رعایت محدودیّت استفاده از دروازه‌های پایه، عمل «یای انحصاری» نیز به‌صورت \(S=\overline{A}B+A\overline{B}\) با دروازه‌های «و»، «یا» و «نقیض» پیاده‌سازی شد؛ بنابراین نماد \(\oplus\) تنها صورت فشردهٔ تابع منطقی است. سپس دو نیم‌جمع‌کننده و یک دروازهٔ «یا» یک جمع‌کنندهٔ کامل را تشکیل دادند. در این مدار، اگر \(S_1=A\oplus B\) و \(C_1=AB\) باشند، خروجی‌ها از روابط \(S=S_1\oplus C_{\mathrm{in}}\) و \(C_{\mathrm{out}}=C_1+S_1C_{\mathrm{in}}\) به دست می‌آیند. چهار نمونه از جمع‌کنندهٔ کامل نیز با زنجیره‌کردن مسیر نقلی به یک جمع‌کنندهٔ چهاربیتی متّصل شدند.

	\shot{screenshots/Div3/1.png}{پیاده‌سازی نیم‌جمع‌کننده با دروازه‌های پایه}
	\shot{screenshots/Div3/2.png}{ساخت جمع‌کننده‌ی کامل از دو نیم‌جمع‌کننده}
	\shot{screenshots/Div3/3.png}{جمع‌کنندهٔ چهاربیتی با اتّصال آبشاری چهار جمع‌کنندهٔ کامل}
	\shot{screenshots/Div3/4.png}{شبیه‌سازی جمع‌کنندهٔ چهاربیتی و مشاهدهٔ جمع و نقلی}

	پیمانهٔ \lr{rem} با منطق ترکیبی، باقی‌ماندهٔ یک رقم دهدهیِ دودویی‌شده بر ۳ را تولید می‌کند. باقی‌مانده در \(O[1..0]\) قرار می‌گیرد و دو بیت بالاییِ \(O[3..2]\) همواره صفرند. نگاشت رقم‌های صفر تا ۹ به‌ترتیب دنبالهٔ \(0,1,2,0,1,2,0,1,2,0\) است. ورودی‌های ۱۰ تا ۱۵ نمایش معتبر دهدهیِ دودویی‌شده نیستند و در طراحی و آزمون استفاده نمی‌شوند. در مدار \lr{Div3} باقی‌ماندهٔ چهار رقم ابتدا محاسبه، به کمک سه جمع‌کننده با هم جمع و دوباره بر ۳ تقلیل داده می‌شود. خروجی \lr{div3} تنها هنگامی یک است که دو بیت باقی‌ماندهٔ نهایی صفر باشند.

	\shot{screenshots/Div3/5.png}{منطق دروازه‌ای پیمانهٔ محاسبهٔ باقی‌مانده بر ۳}
	\shot{screenshots/Div3/7.png}{شبیه‌سازی پیمانهٔ باقی‌مانده برای رقم‌های صفر تا ۹}
	\shot{screenshots/Div3/8.png}{طرح‌وارهٔ سلسله‌مراتبی آشکارساز مضرب ۳ چهاررقمی}

	\subsection*{ب) تشخیص مضرب ۱۱}
	از آنجا که \(10\equiv-1\pmod {11}\)، شرط بخش‌پذیری عدد چهاررقمی برابر است با
	\[
		N\bmod11=(D-C+B-A)\bmod11=0,
	\]
	یا به‌صورت معادل، برابری دو جمع متناوب \(A+C\) و \(B+D\) با در نظر گرفتن حالت‌های هم‌نهشت بر ۱۱.

	اگر \(X=A+C\) و \(Y=B+D\) باشد، با توجّه به \(0\leq X,Y\leq18\)، عدد ورودی بر ۱۱ بخش‌پذیر است اگر و تنها اگر یکی از سه شرط \(X=Y\)، \(X=Y+11\) یا \(Y=X+11\) برقرار باشد. برای انجام این بررسی، مقایسه‌کنندهٔ یک‌بیتی با سه خروجی بزرگ‌تر، برابر و کوچک‌تر ساخته شد. دو نمونه از آن یک مقایسه‌کنندهٔ دوبیتی و دو نمونهٔ مقایسه‌کنندهٔ دوبیتی یک مقایسه‌کنندهٔ چهاربیتی را تشکیل می‌دهند؛ در هر سطح، نتیجهٔ بیت پرارزش بر نتیجهٔ بیت کم‌ارزش اولویت دارد. یک چندراه‌گزین چهاربیتی نیز با چهار چندراه‌گزین دروازه‌ای یک‌بیتی پیاده‌سازی شد.

	\shot{screenshots/Div11/1.png}{مقایسه‌کننده‌ی یک‌بیتی با خروجی‌های بزرگ‌تر، برابر و کوچک‌تر}
	\shot{screenshots/Div11/2.png}{ساخت مقایسه‌کننده‌ی دوبیتی از دو مقایسه‌کننده‌ی یک‌بیتی}
	\shot{screenshots/Div11/3.png}{ساخت مقایسه‌کننده‌ی چهاربیتی از دو مقایسه‌کننده‌ی دوبیتی}
	\shot{screenshots/Div11/4.png}{شبیه‌سازی حالت‌های بزرگ‌تر، کوچک‌تر و برابر مقایسه‌کننده}
	\shot{screenshots/Div11/5.png}{چندراه‌گزین چهاربیتی ساخته‌شده از دروازه‌های پایه}

	در مدار نهایی \lr{Div11}، دو جفت رقم متناوب به‌وسیلهٔ جمع‌کننده‌های چهاربیتی جمع می‌شوند. بیت نقلی هر جمع مشخّص می‌کند که حاصل از ۱۵ گذشته است یا نه. شبکهٔ جمع‌کننده‌ها و چندراه‌گزین‌ها گزینه‌های اصلاح‌شدهٔ دو جمع را برای بررسی برابری مستقیم یا اختلاف ۱۱ می‌سازد و مقایسه‌کنندهٔ نهایی سه حالت یادشده را می‌آزماید. خروجی برابری این مقایسه‌کننده به‌عنوان خروجی \lr{Div11} در نظر گرفته می‌شود.

	\shot{screenshots/Div11/6.png}{طرح‌وارهٔ سلسله‌مراتبی آشکارساز مضرب ۱۱}

	\subsection*{ج) مدار نهایی}
	دو آشکارساز مستقل در پیمانهٔ \lr{final} قرار گرفتند و چهار رقم دهدهیِ دودویی‌شدهٔ مشترک به هر دو اعمال شد. بنابراین خروجی‌های \lr{is\_Div3} و \lr{is\_Div11} می‌توانند هم‌زمان و بدون اثرگذاری بر یکدیگر وضعیّت بخش‌پذیری عدد ورودی را گزارش کنند.

	\shot{screenshots/Final/1.png}{اتّصال آشکارسازهای مضرب ۳ و مضرب ۱۱ در مدار نهایی}

	\section*{۱.۳ شبیه‌سازی و ارزیابی}
	در شبیه‌سازی \lr{Div3} چند عدد چهاررقمی اعمال شده است. برای نمونه، ارقام \(1,2,3,6\) عدد ۱۲۳۶ را می‌سازند و چون جمع رقم‌ها \(12\) است، خروجی \lr{div3} یک می‌شود. برای ۲۳۴۱ با جمع رقم‌های \(10\) خروجی صفر، برای ۳۱۰۵ با جمع رقم‌های \(9\) خروجی یک و برای ۵۰۳۸ با جمع رقم‌های \(16\) خروجی دوباره صفر است. این رفتار با معیار نظری بخش‌پذیری بر ۳ سازگار است.

	\shot{screenshots/Div3/9.png}{شبیه‌سازی آشکارساز مضرب ۳ برای چند عدد چهاررقمی}

	شکل موج \lr{Div11} حالت‌های گوناگون دو جمع متناوب را می‌آزماید. برای ۰۰۰۰، دو جمع برابر صفرند و خروجی یک است؛ برای ۱۲۳۲ نیز \(A+C=B+D=4\) و خروجی یک می‌شود. بردار ۱۲۳۴، با دو جمع ۴ و ۶، نمونهٔ اختلاف نامعتبر و دارای خروجی صفر است. بردار ۹۰۲۰ حالت \(X=Y+11\)، بردارهای ۹۷۹۰ و ۷۹۰۹ حالت \(Y=X+11\)، و بردار ۹۹۹۹ حالت برابری دو جمع را می‌آزمایند؛ در همهٔ این حالت‌ها خروجی یک است. در پایان، ۹۹۹۸ با اختلاف یک، خروجی صفر تولید می‌کند. بدین‌ترتیب، برابری، هر دو جهت اختلاف ۱۱ و یک اختلاف نامعتبر پوشش داده شده‌اند.

	\shot{screenshots/Div11/7.png}{شبیه‌سازی آشکارساز مضرب ۱۱ برای بردارهای آزمون}

	در آزمون مدار نهایی، هر چهار رقم به هر دو زیرمدار داده شده‌اند. شکل موج نشان می‌دهد که دو خروجی مستقل‌اند: برای ۳۶۶۳ هر دو شرط برقرارند؛ برای ۱۱۱۲ هیچ‌کدام برقرار نیست؛ برای ۵۷۶۳ فقط شرط مضرب ۳ و برای ۶۱۲۷ فقط شرط مضرب ۱۱ برقرار است. این چهار حالت، ترکیب‌های ممکن دو خروجی را پوشش می‌دهند.

	\shot{screenshots/Final/3.png}{شبیه‌سازی یکپارچه و پوشش چهار حالت خروجی دو آشکارساز}

	\section*{۱.۵ نتایج}
	\begin{itemize}
		\item زیرمدارهای نیم‌جمع‌کننده، جمع‌کنندهٔ کامل، جمع‌کنندهٔ چهاربیتی، مقایسه‌کننده‌ها و چندراه‌گزین تنها با طرح‌واره و دروازه‌های پایه ساخته شدند.
		\item معیار جمع رقم‌ها برای تشخیص مضرب ۳ به‌صورت سلسله‌مراتبی پیاده‌سازی و در شبیه‌سازی تأیید شد.
		\item معیار اختلاف جمع رقم‌های متناوب برای تشخیص مضرب ۱۱ با جمع‌کننده، چندراه‌گزین و مقایسه‌کننده پیاده‌سازی شد.
		\item شبیه‌سازی مدار نهایی هر چهار ترکیب ممکن خروجی‌های \lr{is\_Div3} و \lr{is\_Div11} را نشان داد.
		\item هر سه طرح بدون استفاده از ثبات و با مصرف کمتر از یک درصد عناصر منطقی قطعه سنتز شدند.
	\end{itemize}

	\section*{نتیجه‌گیری}
	در این آزمایش دو مدار ترکیبی تشخیص بخش‌پذیری برای ورودی چهاررقمی با رمزگذاری دهدهیِ دودویی‌شده، به روش طراحی طرح‌واره‌ای ساخته شدند. شکستن طرح به پیمانه‌های کوچک و قابل‌آزمون، از نیم‌جمع‌کننده و مقایسه‌کنندهٔ یک‌بیتی تا آشکارسازهای کامل، بررسی مرحله‌به‌مرحلهٔ عملکرد را ممکن کرد. شکل موج‌ها با قواعد ریاضی بخش‌پذیری بر ۳ و ۱۱ سازگارند و گزارش‌های سنتز و نماهای پس از نگاشت و جانمایی نیز پیاده‌سازی موفّق و کم‌هزینهٔ طرح روی قطعهٔ \lr{Cyclone IV GX} را تأیید می‌کنند.
