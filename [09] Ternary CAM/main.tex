	\section*{۹.۱ مقدمه و اهداف}
	هدف این آزمایش، طراحی و پیاده‌سازی یک حافظهٔ شرکت‌پذیر\LTRfootnote{Content Addressable Memory (CAM)} نوع سه‌گانه\LTRfootnote{Ternary CAM (TCAM)} است. مطابق دستورکار، این پیمانه باید به اندازهٔ ۱۶ ثبات ۱۶ بیتی باشد.

	تفاوت بنیادی این حافظه با حافظهٔ متعارف در جهت آدرس‌دهی است. در حافظهٔ معمولی، آدرس داده می‌شود و محتوا بازگردانده می‌شود؛ در حافظهٔ شرکت‌پذیر عکس این اتّفاق می‌افتد: محتوا داده می‌شود و آدرسی که آن محتوا در آن ذخیره شده بازگردانده می‌گردد. آنچه نوع سه‌گانه را از حافظهٔ شرکت‌پذیر عادی جدا می‌کند، این است که در هر خانه افزون بر مقادیر صفر و یک، می‌توان مقدار سوم \lr{X} را نیز ذخیره کرد. دستورکار معنای این مقدار را روشن کرده است: مقایسه تنها برای جایگاه‌هایی انجام می‌شود که صفر یا یک‌اند و در جایگاه \lr{X} مقایسه‌ای صورت نمی‌گیرد و بیت‌ها برابر فرض می‌شوند.

	دستورکار برای روشن شدن این رفتار مثالی نیز آورده است: اگر دادهٔ \(01101110\) در چنین حافظه‌ای جست‌وجو شود، با هر یک از الگوهای \(0110\mathrm{XXXX}\)، \(0\mathrm{X}1\mathrm{X}11\mathrm{X}0\) و \(\mathrm{X}1101\mathrm{XXX}\) به‌عنوان انطباق\LTRfootnote{Match} در نظر گرفته می‌شود. این مثال در بخش شبیه‌سازی همین گزارش، بیت‌به‌بیت بازسازی و آزموده شده است.

	\subsection*{روند انجام آزمایش}
	ابتدا شیوهٔ نمایش مقدار سه‌گانه در سخت‌افزار دودویی تعیین و رابطهٔ منطقی جست‌وجو استخراج شد. سپس پیمانهٔ \lr{TCAM} به‌صورت پارامتری نوشته شد تا هم پیکربندی خواسته‌شدهٔ دستورکار را بپوشاند و هم به آن وابسته نماند. برای ارزیابی، دو میز آزمون\LTRfootnote{Testbench} خودآزما\LTRfootnote{Self-Checking Testbench} ساخته شد: یکی برای پیکربندی پیش‌فرض که هفت سناریو از جمله مثال خودِ دستورکار را می‌آزماید، و دیگری برای پیکربندی متفاوت که پارامتری بودن واقعی طرح را نشان می‌دهد. پس از تأیید در \lr{ModelSim}، طرح در \lr{Quartus II} سنتز و با یک پروندهٔ قیود طراحی تحلیل زمانی شد.

	\section*{۹.۲ نمایش مقدار سه‌گانه و رابطه‌ی جست‌وجو}
	\subsection*{الف) چرا هر خانه به دو ثبات نیاز دارد}
	سخت‌افزار دودویی نمی‌تواند سه مقدار را در یک بیت نگه دارد. بنابراین هر خانهٔ \lr{TCAM} با دو ثبات هم‌عرض نمایش داده می‌شود: یک ثبات داده و یک ثبات نقاب\LTRfootnote{Mask}. قرارداد به‌کاررفته در این طرح چنین است:
	\begin{itemize}
		\item بیت نقاب برابر صفر: بیت متناظر معنادار است و باید با بیت کلید جست‌وجو برابر باشد.
		\item بیت نقاب برابر یک: بیت متناظر بی‌اهمّیّت\LTRfootnote{Don't-Care} است و همان مقدار \lr{X} دستورکار را نمایندگی می‌کند.
	\end{itemize}
	با این قرارداد، هر جایگاه سه‌گانه با یک زوج \((\text{داده},\text{نقاب})\) کد می‌شود و مقدار بیت داده در جایگاه‌های نقاب‌شده بی‌اثر است.

		\centering

			\hline
			\textbf{بیت نقاب} & \textbf{بیت داده} & \textbf{مقدار ذخیره‌شده} & \textbf{رفتار در مقایسه} \\ \hline
			\(0\) & \(0\) & \(0\) & باید با بیت کلید برابر باشد \\ \hline
			\(0\) & \(1\) & \(1\) & باید با بیت کلید برابر باشد \\ \hline
			\(1\) & \(-\) & \lr{X} & هر مقداری در کلید پذیرفته می‌شود \\ \hline
		\end{tabular}

	\end{table}

	\subsection*{ب) رابطه‌ی جست‌وجو}
	با این کدگذاری، شرط انطباق یک خانه با کلید جست‌وجو به یک رابطهٔ بیتی ساده تبدیل می‌شود. برای خانهٔ \(i\)ام:
	\[
	\text{انطباق}_i \;=\; \Bigl(\bigl(\text{کلید} \oplus \text{داده}_i\bigr) \;\wedge\; \overline{\text{نقاب}_i}\Bigr) = 0 .
	\]
	منطق این رابطه گام‌به‌گام چنین است. یای‌انحصاری میان کلید و داده، در هر جایگاهی که این دو با هم فرق دارند مقدار یک می‌دهد؛ یعنی نقشهٔ اختلاف‌ها را می‌سازد. واو کردن این نقشه با نقیض نقاب، اختلاف‌های واقع در جایگاه‌های بی‌اهمّیّت را حذف می‌کند و تنها اختلاف‌های واقع در جایگاه‌های معنادار را باقی می‌گذارد. سرانجام، صفر بودن کلّ نتیجه یعنی هیچ اختلاف معناداری وجود ندارد و خانه با کلید منطبق است.

	برای نمونه، مثال خودِ دستورکار را با این رابطه بررسی می‌کنیم. کلید \(01101110\) و الگوی \(0110\mathrm{XXXX}\) را در نظر بگیرید که با داده \(01100000\) و نقاب \(00001111\) کد می‌شود:
	\[
	\begin{aligned}
	\text{کلید} \oplus \text{داده} &= 01101110 \oplus 01100000 = 00001110,\\
	\overline{\text{نقاب}} &= 11110000,\\
	00001110 \wedge 11110000 &= 00000000 \;\Rightarrow\; \text{انطباق}.
	\end{aligned}
	\]
	اختلاف‌ها همگی در چهار جایگاه پایین رخ داده‌اند که دقیقاً همان جایگاه‌های نقاب‌شده‌اند، پس نادیده گرفته می‌شوند.

	\subsection*{ج) بیت اعتبار}
	یک نکتهٔ ظریف در همین رابطه نهفته است: خانه‌ای که هرگز نوشته نشده و محتوای آن صفر است، در برابر کلید صفر انطباق نشان می‌دهد. بدتر از آن، خانه‌ای با نقاب سراسر یک با \emph{هر} کلیدی منطبق می‌شود. برای جلوگیری از این انطباق‌های کاذب، به ازای هر خانه یک بیت اعتبار\LTRfootnote{Valid Bit} نگه داشته می‌شود که تنها با نوشتن در آن خانه یک می‌شود و با بازنشانی صفر می‌گردد. شرط نهایی انطباق، واو منطقی این بیت با رابطهٔ بالاست.

	\subsection*{ب) جست‌وجوی موازی و اولویّت آدرس}
	جست‌وجو در یک بلوک ترکیبی انجام می‌شود که با یک حلقهٔ \lr{for} روی همهٔ خانه‌ها می‌گردد. این حلقه در زمان سنتز باز می‌شود و به شانزده مقایسه‌کنندهٔ مستقل تبدیل می‌گردد که همگی هم‌زمان کار می‌کنند؛ یعنی جست‌وجو ذاتاً موازی است و زمان آن به شمار خانه‌ها بستگی ندارد. همین ویژگی، دلیل اصلی استفاده از این نوع حافظه در کاربردهایی مانند جدول مسیریابی است.

	وقتی بیش از یک خانه منطبق شود، باید تصمیمی دربارهٔ خروجی تک‌مقداری گرفت. در این طرح، چون حلقه از آدرس کوچک به بزرگ پیش می‌رود و هر انطباق تازه مقدار پیشین را بازنویسی می‌کند، آدرس \emph{بزرگ‌ترین} خانهٔ منطبق در خروجی می‌ماند. در مقابل، \lr{match\_vector} هیچ اطّلاعاتی را دور نمی‌ریزد و همهٔ انطباق‌ها را گزارش می‌کند. این تفکیک عمدی است: خروجی تک‌مقداری برای کاربرد متعارف و بردار برای زمانی که شمار یا مجموعهٔ انطباق‌ها اهمّیّت دارد.

	\subsection*{ج) نوشتن، جست‌وجو و اولویّت میان آن‌ها}
	بلوک هم‌زمان طرح سه شاخه دارد. بازنشانی ناهم‌زمان\LTRfootnote{Asynchronous Reset} همهٔ بیت‌های اعتبار و خروجی‌ها را صفر می‌کند. اگر اجازهٔ نوشتن فعّال باشد، داده و نقاب در خانهٔ آدرس‌داده‌شده نوشته و بیت اعتبار آن یک می‌شود. تنها در صورتی که نوشتن فعّال نباشد و اجازهٔ جست‌وجو فعّال باشد، نتیجهٔ بلوک ترکیبی در ثبات‌های خروجی ثبت می‌گردد.

	ترتیب این شاخه‌ها یعنی نوشتن بر جست‌وجو اولویّت دارد. این انتخاب از ابهام جلوگیری می‌کند: اگر هر دو اجازه هم‌زمان فعّال شوند، روشن است که در آن پالس ساعت نوشتن انجام می‌شود و خروجی‌های جست‌وجو دست‌نخورده می‌مانند. همچنین توجّه شود که خروجی‌ها ثبت‌شده‌اند و نه ترکیبی؛ یعنی نتیجهٔ جست‌وجو یک پالس ساعت پس از اعمال کلید در خروجی ظاهر می‌شود و تا جست‌وجوی بعدی پایدار می‌ماند.

	وارسی \lr{addr\_in < DEPTH} نیز عمدی است و تنها هنگامی اهمّیّت پیدا می‌کند که عمق حافظه توانی از دو نباشد؛ در چنین حالتی، پهنای آدرس بیش از عمق واقعی مقدار می‌پذیرد و بدون این وارسی، نوشتن در آدرس خارج از محدوده رفتار نامعیّن می‌ساخت.

	\subsection*{ج) بازه‌ی نخست: نقاب‌های متعارف}

	\subsection*{ه) بازه‌ی سوم: مثال خودِ دستورکار}
	\shot{screenshots/wave_zoom_3.png}{نمای بزرگ‌شده‌ی بازه‌ی مثال دستورکار و انباشت سه انطباق}

	این بازه مهم‌ترین بخش شبیه‌سازی است، چون دقیقاً همان مثالی را بازسازی می‌کند که دستورکار آورده بود. سه الگوی خواسته‌شده به‌ترتیب در خانه‌های \(6\)، \(7\) و \(8\) نوشته می‌شوند و پس از هر نوشتن، همان کلید \(01101110\) جست‌وجو می‌شود. جدول زیر کدگذاری هر الگو و بردار انطباق حاصل را می‌آورد.

	\begin{table}[H]
		\centering
		\footnotesize
		\begin{tabular}{|c|c|c|c|c|c|}
			\hline
			\textbf{زمان (ns)} & \textbf{خانه} & \textbf{الگو} & \textbf{داده} & \textbf{نقاب} & \textbf{بردار انطباق} \\ \hline
			\(135\) & \(6\) & \(0110\mathrm{XXXX}\)      & \(01100000\) & \(00001111\) & خانهٔ \(6\) \\ \hline
			\(155\) & \(7\) & \(\mathrm{X}1101\mathrm{XXX}\) & \(01101000\) & \(10000111\) & خانه‌های \(6,7\) \\ \hline
			\(175\) & \(8\) & \(0\mathrm{X}1\mathrm{X}11\mathrm{X}0\) & \(00101100\) & \(01010010\) & خانه‌های \(6,7,8\) \\ \hline
		\end{tabular}
		\caption{بازسازی مثال دستورکار؛ سه الگوی \lr{X}دار در برابر کلید \(01101110\)}
	\end{table}

	درستی هر سه انطباق را می‌توان دستی وارسی کرد. برای الگوی دوم:
	\[
	01101110 \oplus 01101000 = 00000110,
	\qquad
	00000110 \wedge \overline{10000111} = 00000110 \wedge 01111000 = 0 ,
	\]
	و برای الگوی سوم:
	\[
	01101110 \oplus 00101100 = 01000010,
	\qquad
	01000010 \wedge \overline{01010010} = 01000010 \wedge 10101101 = 0 .
	\]
	در هر دو حالت، همهٔ اختلاف‌ها در جایگاه‌های نقاب‌شده قرار گرفته‌اند و انطباق برقرار است.

	آنچه در شکل موج بیش از همه دیدنی است، انباشته شدن بردار انطباق در سه جست‌وجوی پیاپی است: نخست تنها بیت ششم، سپس بیت‌های ششم و هفتم، و سرانجام هر سه بیت ششم تا هشتم بالا می‌روند. این همان چیزی است که دستورکار می‌گوید: هر سه الگو با آن کلید منطبق‌اند. هم‌زمان، \lr{matched\_addr} در سه لحظه به‌ترتیب مقادیر \(6\)، \(7\) و \(8\) را می‌گیرد، یعنی همواره بزرگ‌ترین آدرس منطبق را گزارش می‌کند.

	\subsection*{و) بازه‌ی چهارم: انطباق فراگیر و بازنویسی}
	\shot{screenshots/wave_zoom_4.png}{نمای بزرگ‌شده‌ی بازه‌ی پایانی؛ خانه‌ی سراسر \lr{X}، انطباق چندگانه و بازنویسی}

	در لحظهٔ ۱۸۵ نانوثانیه خانهٔ \(4\) با نقاب \(FFFF\) نوشته می‌شود، یعنی خانه‌ای که همهٔ شانزده جایگاه آن بی‌اهمّیّت است. چنین خانه‌ای با هر کلیدی منطبق می‌شود و جست‌وجوی کلید دلخواه \(BEEE\) در لحظهٔ ۱۹۵ همین را تأیید می‌کند. این حالت مرزی، هم رفتار درست نقاب کامل را می‌سنجد و هم یادآور می‌شود که چرا بیت اعتبار لازم است: پیش از نوشتن، همین خانه نباید با هیچ کلیدی منطبق می‌شد.

	در لحظهٔ ۲۰۵ خانهٔ \(5\) نیز با نقاب سراسر یک نوشته می‌شود. اکنون دو خانهٔ فراگیر وجود دارد و جست‌وجوی کلید \(CAFE\) در لحظهٔ ۲۱۵ هر دو را روشن می‌کند؛ بردار انطباق مقدار \(0030_{16}\) می‌گیرد که بیت‌های چهارم و پنجم آن یک است، و \lr{matched\_addr} مقدار \(5\) را نشان می‌دهد، یعنی بزرگ‌ترین آدرس.

	سرانجام در لحظهٔ ۲۲۵، خانهٔ \(5\) با داده \(CAFE\) و نقاب صفر بازنویسی می‌شود. جست‌وجوی بعدی در لحظهٔ ۲۳۵ نکتهٔ ظریفی را آشکار می‌کند: بردار انطباق همچنان همان \(0030_{16}\) است، ولی به دلیلی متفاوت. پیش از بازنویسی، خانهٔ \(5\) به‌سبب نقاب فراگیر با \emph{هر} کلیدی منطبق می‌شد؛ پس از بازنویسی، تنها به این دلیل منطبق است که محتوای آن دقیقاً برابر همان کلید \(CAFE\) است. خروجی \lr{matched\_num} این تفاوت را آشکار می‌کند و از \(FFFF\) به \(CAFE\) تغییر می‌یابد. این وارسی نشان می‌دهد که بازنویسی واقعاً محتوای پیشین را جایگزین کرده است و نه اینکه خانهٔ تازه‌ای ساخته باشد.

	\subsection*{ز) پیکربندی پارامتری}
	\shot{screenshots/wave_param.png}{شکل موج اجرای میز آزمون پارامتری با پهنای ۸ بیت و عمق ۵ خانه}

	در این اجرا، همان پیمانه با پهنای داده هشت بیت و عمق پنج خانه ساخته شده است. پهنای آدرس به‌طور خودکار سه بیت محاسبه می‌شود و بردار انطباق نیز به پنج بیت کوتاه می‌شود؛ هر دو در شکل موج قابل مشاهده‌اند. سه وارسی نخست همان رفتارهای نقاب‌دار، ناموفّق و دقیق را در مقیاس کوچک‌تر تکرار می‌کنند.

	وارسی چهارم مخصوص همین پیکربندی است. آدرس \(7\) روی گذرگاه سه‌بیتی گذاشته می‌شود، حال آنکه حافظه تنها پنج خانه با آدرس‌های صفر تا چهار دارد. در شکل موج دیده می‌شود که با وجود فعّال بودن اجازهٔ نوشتن، هیچ بیت اعتباری تغییر نمی‌کند و جست‌وجوی بعدی نیز بدون انطباق می‌ماند. این همان وارسی \lr{addr\_in < DEPTH} است که در بند ۹.۳ توضیح داده شد و اهمّیّت آن تنها در عمق‌های غیرتوان دو آشکار می‌شود.

	\subsection*{ح) جمع‌بندی تحلیل}
	\begin{itemize}
		\item هر رویداد شبیه‌سازی دقیقاً یک پالس ساعت طول می‌کشد و کلّ اجرا زنجیره‌ای منظّم از لحظه‌های ۲۵ تا ۲۳۵ نانوثانیه است.
		\item بیت اعتبار در نخستین جست‌وجو، پیش از هر نوشتنی، از انطباق کاذب جلوگیری کرد.
		\item مثال خودِ دستورکار با هر سه الگوی \lr{X}دار بازسازی شد و بردار انطباق در سه گام پیاپی هر سه خانه را روشن کرد.
		\item در همهٔ حالت‌های انطباق چندگانه، خروجی تک‌مقداری بزرگ‌ترین آدرس را داد و بردار، همهٔ انطباق‌ها را نگه داشت.
		\item بازنویسی خانه، محتوای پیشین را واقعاً جایگزین کرد؛ تفاوت آن در تغییر \lr{matched\_num} از \(FFFF\) به \(CAFE\) دیده شد.
		\item پیکربندی پارامتری بدون هیچ تغییری در کد اجرا شد و وارسی آدرس خارج از محدوده نیز درست عمل کرد.
	\end{itemize}

	\section*{۹.۷ سنتز و تحلیل زمانی}
	\subsection*{الف) مصرف منابع}
	طرح در \lr{Quartus II 13.0.1} و برای خانوادهٔ \lr{Cyclone IV GX} همگردانی شد. انتخاب قطعه روی حالت خودکار بود و ابزار قطعهٔ \lr{EP4CGX22CF19C6} را برگزید. نتیجه ۶۵۴ عنصر منطقی شامل ۵۴۶ تابع ترکیبی و ۵۶۵ ثبات اختصاصی، و ۷۷ پایه است.

	\shotw{screenshots/flow_summary.png}{0.72}{خلاصه‌ی منابع مصرفی پیمانه‌ی \lr{TCAM}}

	هر دو عدد کلیدی این گزارش را می‌توان مستقیماً از روی کد بازسازی کرد. شمار ثبات‌ها برابر است با آرایهٔ داده، آرایهٔ نقاب، بیت‌های اعتبار و چهار ثبات خروجی:
	\[
	\underbrace{16\times16}_{\text{داده}} + \underbrace{16\times16}_{\text{نقاب}} + \underbrace{16}_{\text{اعتبار}} + \underbrace{16+4+16+1}_{\text{خروجی‌ها}} = 565 ,
	\]
	که دقیقاً با عدد گزارش‌شده می‌خواند. شمار پایه‌ها نیز برابر است با چهل پایهٔ ورودی و سی و هفت پایهٔ خروجی:
	\[
	\underbrace{1+1+1+1+16+16+4}_{40} \;+\; \underbrace{16+4+16+1}_{37} = 77 .
	\]

	نکتهٔ درخور توجّه آن است که مقدار بیت‌های حافظهٔ مصرف‌شده صفر گزارش شده است. یعنی سنتز‌کننده هیچ بلوک حافظهٔ آماده‌ای را به کار نگرفته و کلّ آرایهٔ ۵۱۲ بیتی را با ثبات‌های معمولی ساخته است. این نتیجه اجتناب‌ناپذیر است و ماهیّت \lr{TCAM} را نشان می‌دهد: بلوک‌های حافظهٔ آماده تنها یک خانه را در هر دسترسی می‌خوانند، حال آنکه اینجا باید محتوای همهٔ خانه‌ها هم‌زمان و به‌صورت موازی با کلید مقایسه شود. همین نیاز به دسترسی هم‌زمان است که \lr{TCAM} را نسبت به حافظهٔ معمولی پرهزینه می‌کند.

	\subsection*{ج) تحلیل زمانی}

	تحلیل‌گر زمانی\LTRfootnote{TimeQuest Timing Analyzer} در بدترین شرایط کاری حاشیهٔ برپایی\LTRfootnote{Slack} \(0.758\) نانوثانیه و حاشیهٔ نگهداری \(0.646\) نانوثانیه گزارش می‌کند و مجموع حاشیه‌های منفی در همهٔ تحلیل‌ها صفر است. بیشینه‌بسامد\LTRfootnote{Maximum Frequency (Fmax)} نیز \(108.2\,\mathrm{MHz}\) است و بیشینه‌بسامد مقیّد همین مقدار است؛ یعنی محدودکنندهٔ سرعت، منطق درونی طرح است و نه نرخ تغییر پایه‌ها.

	\shotw{screenshots/fmax_summary.png}{0.72}{گزارش بیشینه‌بسامد پیمانه‌ی \lr{TCAM}}

	این حاشیه از همهٔ آزمایش‌های پیشین تنگ‌تر است و دلیل آن در ماهیّت طرح نهفته است. مسیر بحرانی از ورودی کلید آغاز می‌شود، از یای‌انحصاری و نقاب شانزده خانه می‌گذرد، سپس باید نتیجهٔ هر خانه در یک درخت آشکارساز صفر شانزده‌بیتی کاهش یابد و سرانجام از منطق اولویّت انتخاب بزرگ‌ترین آدرس عبور کند. این زنجیرهٔ ترکیبی طولانی، بهای همان جست‌وجوی موازی است که مزیّت اصلی \lr{TCAM} به شمار می‌رود. بااین‌حال قید ۱۰۰ مگاهرتزی برآورده شده و طرح با حاشیهٔ مثبت بسته می‌شود.

	با توجّه به برگزاری مجازی آزمایشگاه، ارزیابی این آزمایش بر پایهٔ شبیه‌سازی، سنتز و تحلیل زمانی نرم‌افزاری انجام شده و در پرونده‌های پیکربندی پروژه تخصیص پایه‌ای برای برد ثبت نشده است؛ ازاین‌رو ادّعایی دربارهٔ آزمون فیزیکی روی برد مطرح نمی‌شود.

	\section*{۹.۸ نتایج}
	\begin{itemize}
		\item پیمانهٔ \lr{TCAM} با همان اندازهٔ خواسته‌شدهٔ دستورکار، یعنی ۱۶ خانهٔ ۱۶ بیتی، پیاده‌سازی شد.
		\item مقدار سه‌گانه با زوج داده و نقاب کد شد و شرط انطباق به یک رابطهٔ بیتی ساده تبدیل گردید.
		\item مثال خودِ دستورکار با هر سه الگوی \lr{X}دار بازسازی و در شبیه‌سازی تأیید شد؛ هر سه با کلید \(01101110\) منطبق شدند.
		\item بیت اعتبار از انطباق کاذب خانه‌های نانوشته جلوگیری کرد و این رفتار جداگانه آزموده شد.
		\item جست‌وجو موازی است و زمان آن مستقل از شمار خانه‌هاست؛ در همهٔ حالت‌ها نتیجه در یک پالس ساعت آماده شد.
		\item در انطباق چندگانه، خروجی تک‌مقداری بزرگ‌ترین آدرس را داد و بردار انطباق همهٔ خانه‌های منطبق را گزارش کرد.
		\item طرح با پیکربندی هشت‌بیتی و عمق پنج نیز بدون تغییر در کد اجرا شد و وارسی آدرس خارج از محدوده درست عمل کرد.
		\item سنتز با ۶۵۴ عنصر منطقی، ۵۶۵ ثبات و ۷۷ پایه انجام شد؛ هر دو عدد ثبات و پایه دقیقاً با محاسبهٔ نظری از روی کد می‌خوانند.
		\item تحلیل زمانی با قید ۱۰۰ مگاهرتزی حاشیهٔ مثبت \(0.758\) نانوثانیه و بیشینه‌بسامد \(108.2\,\mathrm{MHz}\) داد.
	\end{itemize}

	\section*{نتیجه‌گیری}
	در این آزمایش یک حافظهٔ شرکت‌پذیر سه‌گانه ساخته شد که به‌جای آدرس، با محتوا پرس‌وجو می‌شود. کلید طراحی، تبدیل مفهوم انتزاعی «بیت بی‌اهمّیّت» به یک زوج داده و نقاب بود؛ تبدیلی که شرط انطباق را به یک رابطهٔ بیتی سه‌عملگری فرو کاست و کلّ جست‌وجو را به منطق ترکیبی ساده تبدیل کرد. در کنار آن، دو جزئیّات کوچک اما تعیین‌کننده بودند: بیت اعتبار که بدون آن خانه‌های نانوشته انطباق کاذب می‌ساختند، و تفکیک خروجی تک‌مقداری از بردار انطباق که بدون آن اطّلاعات انطباق‌های هم‌زمان از دست می‌رفت. سرانجام، گزارش سنتز یادآور شد که مزیّت این حافظه رایگان نیست: صفر بودن بیت‌های حافظهٔ مصرف‌شده نشان داد که کلّ آرایه ناگزیر با ثبات ساخته می‌شود، و تنگ بودن حاشیهٔ زمانی نشان داد که جست‌وجوی موازی، مسیر ترکیبی بلندی می‌سازد.
