	\section*{۳.۱ مقدمه و اهداف}
	در این آزمایش، توصیف جریان داده در \lr{Verilog} با دستور \lr{assign} بررسی می‌شود. بخش نخست شامل طراحی سلسله‌مراتبی مقایسه‌کننده‌ی آبشاری چهار بیتی و بخش دوم شامل طراحی یک مقایسه‌کننده‌ی سریال دارای ورودی \lr{reset} است.

	\section*{۳.۲ طراحی و کد \lr{Verilog}}
	\subsection*{الف) مقایسه‌کننده‌ی آبشاری چهار بیتی}
	واحد پایه، یک مقایسه‌کننده‌ی یک‌بیتی آبشاری با سه ورودی وضعیت از مرحله‌ی پرارزش‌تر است. اگر مراحل قبلی برابری را گزارش کرده باشند، بیت جاری می‌تواند نتیجه را به بزرگ‌تر یا کوچک‌تر تغییر دهد؛ در غیر این صورت نتیجه‌ی قطعی مرحله‌ی قبلی بدون تغییر عبور می‌کند. روابط داده‌ای این مرحله عبارت‌اند از:
	\[
	\begin{aligned}
	G_{\mathrm{next}}&=G_{\mathrm{prev}}\lor(E_{\mathrm{prev}}\land A\land\overline{B}),\\
	E_{\mathrm{next}}&=E_{\mathrm{prev}}\land\overline{A\oplus B},\\
	L_{\mathrm{next}}&=L_{\mathrm{prev}}\lor(E_{\mathrm{prev}}\land\overline{A}\land B).
	\end{aligned}
	\]
	هر سه رابطه تنها با دستورهای پیوسته‌ی \lr{assign} توصیف شده‌اند.

	\shot{screenshots/Cascadable_Comparator/module_compare_stage_code.png}{کد جریان داده‌ی مرحله‌ی یک‌بیتی مقایسه‌کننده}

	برای ساخت مقایسه‌کننده‌ی چهار بیتی، چهار نمونه از این مرحله از بیت ۳ تا بیت ۰ به‌صورت آبشاری متصل شدند. وضعیت اولیه‌ی مرحله‌ی پرارزش \(G=0,E=1,L=0\) است و خروجی هر مرحله به ورودی وضعیت مرحله‌ی بعد می‌رود. در نتیجه، نخستین اختلاف از سمت بیت پرارزش نتیجه‌ی نهایی را تعیین می‌کند.

	\shot{screenshots/Cascadable_Comparator/topfour_bit_comparator_code_1.png}{تعریف ورودی‌ها، خروجی‌ها و مرحله‌ی بیت ۳}
	\shot{screenshots/Cascadable_Comparator/topfour_bit_comparator_code_2.png}{اتصال مرحله‌ی بیت ۳ به مرحله‌ی بیت ۲}
	\shot{screenshots/Cascadable_Comparator/topfour_bit_comparator_code_3.png}{اتصال مراحل بیت ۱ و صفر و تولید خروجی نهایی}

	نمای \lr{RTL} دقیقاً چهار بلوک \lr{compare\_stage} متوالی را نشان می‌دهد و بنابراین ساختار سلسله‌مراتبی موردنظر دستور کار حفظ شده است.

	\shot{screenshots/Cascadable_Comparator/RTL_photo.png}{نمای \lr{RTL} مقایسه‌کننده‌ی آبشاری چهار بیتی}

	\subsection*{ب) مقایسه‌کننده‌ی سریال}
	مقایسه‌کننده‌ی سریال پس از \lr{reset} در حالت برابر، یعنی \(G=0,E=1,L=0\)، قرار می‌گیرد. بیت‌های دو عدد از پرارزش‌ترین بیت به کم‌ارزش‌ترین بیت وارد می‌شوند. تا وقتی \(E=1\) است هر بیت جدید می‌تواند رابطه را تعیین کند؛ پس از مشاهده‌ی نخستین اختلاف، وضعیت بزرگ‌تر یا کوچک‌تر حفظ می‌شود. روابط حالت بعد همان روابط مرحله‌ی آبشاری‌اند، اما نتیجه باید بین پالس‌های ساعت نگه‌داری شود.

	برای رعایت محدودیت استفاده‌ی انحصاری از \lr{assign}، نگه‌داری وضعیت با شش شبکه‌ی بازخوردی در دو فاز \lr{master/slave} انجام شده است. در یک سطح ساعت، حالت بعد در سیگنال‌های \lr{master} ذخیره می‌شود و در سطح دیگر به خروجی منتقل می‌گردد. \lr{reset} نیز به‌صورت غیرهم‌زمان مقادیر اولیه را تحمیل می‌کند.

	\shot{screenshots/Serial_Comparator/serail_bit_comparator_code_1.png}{تعریف رابط و روابط حالت بعد مقایسه‌کننده‌ی سریال}

	\section*{۳.۳ تست‌بنچ و شبیه‌سازی}
	\subsection*{الف) آزمون مقایسه‌کننده‌ی آبشاری}

	\shot{screenshots/Cascadable_Comparator/test_bench_code_1.png}{تعریف تست‌بنچ و نمونه‌سازی مقایسه‌کننده‌ی چهار بیتی}

	\section*{۳.۴ سنتز و پیاده‌سازی نرم‌افزاری}
	هر دو طرح برای قطعه‌ی \lr{EP2C5T144C6} از خانواده‌ی \lr{Cyclone II} سنتز شدند. مقایسه‌کننده‌ی آبشاری ۹ تابع ترکیبی و ۱۱ پایه، و مقایسه‌کننده‌ی سریال ۹ تابع ترکیبی و ۷ پایه مصرف می‌کند؛ در هر دو مورد مصرف عناصر منطقی کمتر از یک درصد ظرفیت تراشه است.

	\section*{۳.۵ نتایج}
	\begin{itemize}
		\item مقایسه‌کننده‌ی چهار بیتی با اتصال سلسله‌مراتبی چهار مرحله‌ی یک‌بیتی و صرفاً با توصیف جریان داده ساخته شد.
		\item آزمون فراگیر تمام ۲۵۶ زوج ورودی بدون خطا پایان یافت.
		\item مقایسه‌کننده‌ی سریال نتیجه‌ی مقایسه‌ی پیشوند دریافت‌شده را در هر پالس نگه داشت و پس از نخستین اختلاف، وضعیت نهایی را حفظ کرد.
		\item حالت بازنشانی \(G=0,E=1,L=0\) و هر شش زوج آزمون سریال با موفقیت بررسی شدند.
		\item هر دو پروژه بدون خطای کامپایل و با مصرف کمتر از یک درصد منابع منطقی سنتز شدند.
	\end{itemize}
